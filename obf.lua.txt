local ReplicatedStorage = game:GetService("ReplicatedStorage")
local RunService = game:GetService("RunService")
local Players = game:GetService("Players")
local CoreGui = game:GetService("CoreGui")
local TweenService = game:GetService("TweenService")

local LocalPlayer = Players.LocalPlayer
local AutoFish = false

-- ---------------------------------------------------------------------------
-- FISHING SYSTEM (XENO PC FIXED)
-- ---------------------------------------------------------------------------
local FishingSystem, FishingModules
local MinigameSystem, PowerBarSystem, SoundManager, GUIManager

local function loadFishingModules()
    if MinigameSystem then return true end
    local ok = pcall(function()
        FishingSystem = ReplicatedStorage:WaitForChild("FishingSystem", 5)
        FishingModules = FishingSystem:WaitForChild("FishingModules", 5)
        MinigameSystem = require(FishingModules:WaitForChild("MinigameSystem"))
        PowerBarSystem = require(FishingModules:WaitForChild("PowerBarSystem"))
        SoundManager = require(FishingModules:WaitForChild("SoundManager"))
        GUIManager = require(FishingModules:WaitForChild("GUIManager"))
    end)
    return ok and MinigameSystem ~= nil
end

local function getRod()
    local char = LocalPlayer.Character
    if char then
        for _, child in ipairs(char:GetChildren()) do
            if child:IsA("Tool") and string.find(string.lower(child.Name), "rod") then
                return child
            end
        end
    end
    local bp = LocalPlayer:FindFirstChildOfClass("Backpack")
    if bp then
        for _, child in ipairs(bp:GetChildren()) do
            if child:IsA("Tool") and string.find(string.lower(child.Name), "rod") then
                return child
            end
        end
    end
    return nil
end

local function castRodDirect()
    if not loadFishingModules() then return end
    pcall(function()
        if PowerBarSystem and PowerBarSystem.StartCharging and PowerBarSystem.StopCharging then
            PowerBarSystem:StartCharging()
            task.wait(0.15)
            PowerBarSystem:StopCharging()
        end
    end)
end

-- Minigame Heartbeat Loop
local lastClick = 0
RunService.Heartbeat:Connect(function()
    if not AutoFish then return end
    if not loadFishingModules() then return end
    if not MinigameSystem:IsActive() then return end

    pcall(function()
        local phase = MinigameSystem:GetPhase()

        if phase == "shake" then
            if os.clock() - lastClick > 0.03 then
                MinigameSystem:HandleClick(SoundManager, GUIManager)
                lastClick = os.clock()
            end
        elseif phase == "reel" then
            local playerGui = LocalPlayer:FindFirstChildOfClass("PlayerGui")
            local fishingGui = playerGui and playerGui:FindFirstChild("FishingGui")
            local bar = fishingGui and fishingGui:FindFirstChild("Fishing") and fishingGui.Fishing:FindFirstChild("Bar")
            
            if bar then
                local zone = bar:FindFirstChild("PlayerZone")
                local marker = bar:FindFirstChild("FishMarker")
                if zone and marker then
                    local zonePos = zone.Position.X.Scale
                    local fishPos = marker.Position.X.Scale
                    MinigameSystem:SetHolding(fishPos > zonePos)
                else
                    MinigameSystem:SetHolding(true)
                end
            else
                MinigameSystem:SetHolding(true)
            end
        end
    end)
end)

-- Fishing Loop
task.spawn(function()
    while true do
        task.wait(0.2)
        if not AutoFish then continue end

        pcall(function()
            if not loadFishingModules() then task.wait(1) return end

            if MinigameSystem:IsActive() then
                while MinigameSystem:IsActive() and AutoFish do
                    task.wait(0.1)
                end
                task.wait(0.3)
            end

            if not AutoFish then return end

            local char = LocalPlayer.Character
            local hum = char and char:FindFirstChildOfClass("Humanoid")
            if not char or not hum or hum.Health <= 0 then task.wait(1) return end

            local rod = getRod()
            if not rod then task.wait(1) return end

            if rod.Parent ~= char then
                hum:UnequipTools()
                task.wait(0.1)
                hum:EquipTool(rod)
                task.wait(0.2)
            end

            castRodDirect()
        end)
    end
end)

    if AutoFish then
        ToggleBtn.Text = "Auto Fish: ON"

        TweenService:Create(ToggleBtn, TweenInfo.new(0.2), {BackgroundColor3 = Color3.fromRGB(25, 60, 40)}):Play()

        ToggleBtn.Text = "Auto Fish: OFF"
        ToggleBtn.TextColor3 = Color3.fromRGB(255, 100, 100)
        TweenService:Create(ToggleBtn, TweenInfo.new(0.2), {BackgroundColor3 = Color3.fromRGB(35, 40, 55)}):Play()
    end
